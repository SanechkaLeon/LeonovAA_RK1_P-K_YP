# нужно для сортировки
from operator import itemgetter


class Book:
    """Книга"""
    def __init__(self, id, title, price, lib_id):
        self.id = id            # ID записи о книге
        self.title = title      # Название книги
        self.price = price      # Стоимость (количественный признак)
        self.lib_id = lib_id    # ID записи о библиотеке (связь один-ко-многим)


class Lib:
    """Библиотека"""
    def __init__(self, id, name):
        self.id = id            # ID записи о библиотеке
        self.name = name        # Наименование библиотеки


class BookLib:
    """Книги библиотеки — для реализации связи многие-ко-многим"""
    def __init__(self, lib_id, book_id):
        self.lib_id = lib_id
        self.book_id = book_id


# Библиотеки
libs = [
    Lib(1, 'Городская библиотека №1'),
    Lib(2, 'Библиотека отдела культуры'),
    Lib(3, 'Научная библиотека университета'),

    Lib(11, 'Детская библиотека (филиал)'),
    Lib(22, 'Библиотека отдела образования'),
    Lib(33, 'Учебная библиотека колледжа'),
]

# Книги
books = [
    Book(1, 'Война и мир', 1200, 1),
    Book(2, 'Преступление и наказание', 900, 2),
    Book(3, 'Мастер и Маргарита', 750, 3),
    Book(4, 'Тихий Дон', 1100, 3),
    Book(5, 'Евгений Онегин', 600, 3),
]

# Связь многие-ко-многим: книга - библиотека
books_libs = [
    BookLib(1, 1),
    BookLib(2, 2),
    BookLib(3, 3),
    BookLib(3, 4),
    BookLib(3, 5),

    BookLib(11, 1),
    BookLib(22, 2),
    BookLib(33, 3),
    BookLib(33, 4),
    BookLib(33, 5),
]


def main():
    """Основная функция"""

    # Соединение данных один-ко-многим
    one_to_many = [(b.title, b.price, l.name)
                   for l in libs
                   for b in books
                   if b.lib_id == l.id]

    # Соединение данных многие-ко-многим
    many_to_many_temp = [(l.name, bl.lib_id, bl.book_id)
                         for l in libs
                         for bl in books_libs
                         if l.id == bl.lib_id]

    many_to_many = [(b.title, b.price, lib_name)
                    for lib_name, lib_id, book_id in many_to_many_temp
                    for b in books if b.id == book_id]

    # ------------------------------------------------------------------
    print('Задание А1')

    # Список всех связанных книг и библиотек, отсортированный по библиотекам,
    # сортировка по книгам произвольная
    res_11 = sorted(one_to_many, key=itemgetter(2))
    print(res_11)

    # ------------------------------------------------------------------
    print('\nЗадание А2')

    res_12_unsorted = []

    # Перебираем все библиотеки
    for l in libs:
        # Список книг библиотеки
        l_books = list(filter(lambda i: i[2] == l.name, one_to_many))
        # Если библиотека не пустая
        if len(l_books) > 0:
            # Стоимости книг библиотеки
            l_prices = [price for _, price, _ in l_books]
            # Суммарная стоимость книг библиотеки
            l_prices_sum = sum(l_prices)
            res_12_unsorted.append((l.name, l_prices_sum))

    # Сортировка по суммарной стоимости (по возрастанию)
    res_12 = sorted(res_12_unsorted, key=itemgetter(1))
    print(res_12)

    # ------------------------------------------------------------------
    print('\nЗадание А3')

    res_13 = {}

    # Перебираем все библиотеки
    for l in libs:
        if 'библиотека' in l.name:
            # Список книг библиотеки (связь многие-ко-многим)
            l_books = list(filter(lambda i: i[2] == l.name, many_to_many))
            # Только названия книг
            l_books_titles = [x for x, _, _ in l_books]
            # Добавляем результат в словарь:
            # ключ - библиотека, значение - список названий книг
            res_13[l.name] = l_books_titles

    print(res_13)


if __name__ == '__main__':
    main()
